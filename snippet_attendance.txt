            isManual ? '<i class="fa-solid fa-users manual-icon" title="Manually edited by admin"></i>' : '',
            isPending ? '<i class="fa-solid fa-triangle-exclamation pending-icon" title="Pending"></i>' : ''
        ].filter(Boolean).join('');
